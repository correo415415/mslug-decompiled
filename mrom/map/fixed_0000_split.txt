; fixed_0000_split.txt - how mrom/asm/raw/fixed_0000.s is carved into subsystem files
; start end file  (addresses in hex, end exclusive)
0000 0080 core/vectors_0000.s            ; reset, RST8 YM write, RST30, IRQ/NMI jumps
0080 0100 data/bank_ptr_table_0080.s     ; 18 x 24-bit pointers into the banks
0100 1e00 data/lookup_pages_0100.s       ; page-aligned lookup tables (freq, volume, envelopes...)
1e00 1fe5 core/main_1e00.s               ; init, main loop, IRQ, NMI, bank switch
1fe5 2076 core/command_2050.s            ; channel struct init, command dispatcher
2076 20d3 core/command_tables_2076.s     ; command jump tables + hi dispatcher
20d3 219b core/command_handlers_20d3.s   ; play/stop music/sfx, fade, tempo
219b 2515 channel/envelope_219b.s        ; envelope / LFO / pitch / vibrato helpers + cmd structs
2515 2b43 channel/ssg_2515.s             ; SSG (PSG) channels A/B/C
2b43 3740 channel/fm_2b43.s              ; FM channels 1-4
3740 3fca channel/adpcma_3740.s          ; ADPCM-A channels 1-6
3fca 4aa0 channel/adpcmb_3fca.s          ; ADPCM-B channel + shared note code
4aa0 4b84 channel/chtype_4aa0.s          ; channel-type dispatch tables, key-on, set-note
4b84 4c84 data/note_map_4b84.s           ; 256-byte note/volume map
4c84 5362 seq/sequencer_4c84.s           ; track allocation, music/sfx start, track update
5362 57a1 seq/opcodes_5362.s             ; sequence opcode fetch + C0..DA handlers + note event
57a1 67a1 data/seq_ptr_table_57a1.s      ; 1024 x 24-bit sequence pointers
67a1 8000 data/tail_67a1.s               ; remaining data of the fixed region
